# CWA integration sketch — not runnable or a reference implementation.
# Implement these application-owned helpers; producers run before assembly.
def prepare_request(req):
    # Keep each batch whole: its items and exclusions stay bound to the producer identity
    # your application authenticated. Never flatten items away from that identity or
    # read identity from item.source.
    batches = [bind_identity(authenticate(producer), producer.run(req)) for producer in producers_for(req)]
    snapshot = freeze_snapshot(
        batches=batches,
        capabilities=capability_grant(req),  # Allow-list result for this route and user.
        conflicts=declared_conflict_groups(req),  # schema/conflict_group.schema.json
        scope=request_scope(req),
        assembly_time=req.assembly_time,
        route_policy=load_versioned_policy(req.route),
        profile=load_profile('policy-first-chat/v2'),
        tokenizer=req.tokenizer,
        renderer=req.renderer,
        budget=req.budget,  # input is already net of reserved_output.
    )
    return assemble_snapshot(snapshot)


def assemble_snapshot(snapshot):
    # No model calls, external reads, or ambient clock reads from this point.
    admitted = admit(snapshot)
    resolved = resolve_declared_conflicts(admitted, snapshot)
    fitted = fit_with_rendered_token_counts(resolved, snapshot)
    if fitted.refused:
        return refused_result(fitted.reason, snapshot)  # No payload; trace.result = null.
    payload = render_profile(fitted, snapshot)
    return result_with_trace_and_sha256(payload, fitted, snapshot)
